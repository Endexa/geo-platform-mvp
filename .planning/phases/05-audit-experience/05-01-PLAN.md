**Step 3: Create `frontend/composables/useApiFetch.ts`.**
This is a typed wrapper around Nuxt's `$fetch` that auto-injects the JWT Authorization header from a cookie.

```typescript
export function useApiFetch<T>(path: string, opts: any = {}): Promise<T> {
  const config = useRuntimeConfig()
  const token = useCookie('access_token')

  return $fetch<T>(path, {
    baseURL: `${config.public.apiUrl}/api/v1`,
    headers: {
      ...(token.value ? { Authorization: `Bearer ${token.value}` } : {}),
      ...opts.headers,
    },
    onResponseError({ response }) {
      if (response.status === 401) {
        token.value = null
        navigateTo('/login')
      }
    },
    ...opts,
  })
}
```

Key points:
- Use `useCookie('access_token')` for SSR-safe token storage (even though SPA mode, this is the standard Nuxt pattern).
- Base URL comes from `useRuntimeConfig().public.apiUrl` + `/api/v1` suffix.
- 401 handler clears token and redirects to /login.
- Generic type parameter `T` for typed responses.

**Step 4: Create `frontend/composables/useAuth.ts`.**
This composable manages auth state (login, register, logout, current user, isLoggedIn).

```typescript
interface User {
  id: string
  email: string
  full_name: string | null
  role: string
}

export const useAuth = () => {
  const token = useCookie('access_token')
  const user = useState<User | null>('auth_user', () => null)
  const isLoggedIn = computed(() => !!token.value)

  const login = async (email: string, password: string) => {
    const data = await useApiFetch<{ access_token: string; refresh_token?: string; token_type: string }>('/auth/login', {
      method: 'POST',
      body: { email, password },
    })
    token.value = data.access_token
    await fetchUser()
  }

  const register = async (email: string, password: string, fullName?: string) => {
    const data = await useApiFetch<{ access_token: string; token_type: string }>('/auth/register', {
      method: 'POST',
      body: { email, password, full_name: fullName || null },
    })
    token.value = data.access_token
    await fetchUser()
  }

  const fetchUser = async () => {
    try {
      const data = await useApiFetch<User>('/auth/me')
      user.value = data
    } catch {
      user.value = null
    }
  }

  const logout = () => {
    token.value = null
    user.value = null
    navigateTo('/login')
  }

  return { token, user, isLoggedIn, login, register, logout, fetchUser }
}
```

Key decisions:
- `useState` for user (survives component unmount, shared across app).
- `useCookie` for token (persistent, SSR-safe).
- Login calls `/auth/login` then fetches `/auth/me` for user profile.
- Register calls `/auth/register` then fetches user.
- Logout clears both token and user, navigates to /login.
- Backend auth endpoints: POST /auth/register returns `{ access_token, token_type }`, POST /auth/login returns `{ access_token, refresh_token, token_type }`, GET /auth/me returns `{ id, email, full_name, role }`.

export default defineNuxtRouteMiddleware((to) => {
  const token = useCookie('access_token')
  if (!token.value && to.path !== '/login') {
