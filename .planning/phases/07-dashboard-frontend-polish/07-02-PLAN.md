1. **Create `composables/useBrandSelector.ts`:**
   ```typescript
   export function useBrandSelector() {
     const selectedBrandId = useState<string | null>('selectedBrandId', () => null)
     const brands = useState<BrandOverview[]>('brands', () => [])
     const brandsLoading = ref(false)
     const brandsError = ref<string | null>(null)

     interface BrandOverview {
       brand_id: string
       brand_name: string
       domain: string
       total_audits: number
       latest_score: number | null
     }

     async function loadBrands() {
       brandsLoading.value = true
       brandsError.value = null
       try {
         const data = await useApiFetch<{ brands: BrandOverview[]; total_brands: number }>('/brands/overview')
         brands.value = data.brands
         // Auto-select first brand if none selected
         if (!selectedBrandId.value && data.brands.length > 0) {
           selectedBrandId.value = data.brands[0].brand_id
         }
       } catch (e: any) {
         brandsError.value = e?.message || 'Failed to load brands'
       } finally {
         brandsLoading.value = false
       }
     }

     const selectedBrand = computed(() =>
       brands.value.find(b => b.brand_id === selectedBrandId.value) || null
     )

     function selectBrand(brandId: string | null) {
       selectedBrandId.value = brandId
     }

     return { selectedBrandId, brands, selectedBrand, brandsLoading, brandsError, loadBrands, selectBrand }
   }
   ```
   Use `useState` (Nuxt SSR-safe global state) so brand selection persists across navigations without Pinia.

2. **Wire brand selector into dashboard.vue:**
   - Import and call `useBrandSelector()` in setup
   - Call `loadBrands()` in `onMounted`
   - Add a brand selector UI below the page title, above the audit selector. Show only when `brands.length > 1`:
     ```
     <div v-if="brands.length > 1" class="flex items-center gap-2 flex-wrap">
       <button
         class="px-3 py-1.5 text-sm font-medium rounded-lg transition-colors duration-200 cursor-pointer"
         :class="!selectedBrandId ? 'bg-primary-800 text-white' : 'bg-gray-100 text-gray-700 hover:bg-gray-200'"
         @click="selectBrand(null)"
       >All Sites</button>
       <button
         v-for="brand in brands"
         :key="brand.brand_id"
         class="px-3 py-1.5 text-sm font-medium rounded-lg transition-colors duration-200 cursor-pointer"
         :class="selectedBrandId === brand.brand_id ? 'bg-primary-800 text-white' : 'bg-gray-100 text-gray-700 hover:bg-gray-200'"
         @click="selectBrand(brand.brand_id)"
       >{{ brand.brand_name }}</button>
     </div>
     ```

3. **Filter audits by brand:**

   ```
   const filteredAudits = computed(() => {

   })
   ```
   Use `filteredAudits` for the audit selector dropdown. When brand changes, auto-select the first audit from the filtered list.
