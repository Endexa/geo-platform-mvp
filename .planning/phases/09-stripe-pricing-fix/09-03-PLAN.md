**Create `frontend/composables/useScrollFadeIn.ts`** — IntersectionObserver-based scroll animation. No animation libraries (constraint from research).

```typescript
import { ref, onMounted } from 'vue'

export function useScrollFadeIn() {
  const el = ref<HTMLElement | null>(null)
  const visible = ref(false)

  onMounted(() => {
    if (!el.value) return
    const observer = new IntersectionObserver(
      ([entry]) => {
        if (entry.isIntersecting) {
          visible.value = true
          observer.unobserve(entry.target)
        }
      },
      { threshold: 0.1 }
    )
    observer.observe(el.value)
  })

  return { el, visible }
}
```

Usage pattern in section components: `<div ref="el" :class="['transition-all duration-500', visible ? 'opacity-100 translate-y-0' : 'opacity-0 translate-y-6']">`
