2. Create `backend/app/tasks/code_analysis_tasks.py`:
   ```python
   import asyncio
   from datetime import datetime, timezone
   from loguru import logger
   from app.core.config import settings
   from app.core.database import SessionLocal
   from app.models.code_analysis import CodeAnalysis
   from app.schemas.code_analysis import CodeAnalysisResult
   from app.services.code_analysis_service import CodeAnalysisService
   from app.tasks.celery_app import celery_app

   @celery_app.task(bind=True, name="app.tasks.code_analysis.process_upload", max_retries=1, default_retry_delay=60)
   def process_code_analysis(self, code_analysis_id: str):
       db = SessionLocal()
       try:
           ca = db.query(CodeAnalysis).filter(CodeAnalysis.id == code_analysis_id).first()
           if not ca:
               logger.error(f"CodeAnalysis {code_analysis_id} not found")
               return
           service = CodeAnalysisService()
           ca.status = "parsing"
           db.commit()
           result = asyncio.run(service.parse_output(ca.raw_output))
           ca.parsed_findings = result.model_dump()
           ca.detected_framework = result.detected_framework
           ca.model_used = settings.CODE_ANALYSIS_MODEL
           ca.status = "recommending"
           db.commit()
           service.generate_code_recommendations(ca, result, db)
           ca.status = "completed"
           ca.completed_at = datetime.now(timezone.utc)
           db.commit()
           logger.info(f"Code analysis {code_analysis_id} completed with {len(result.findings)} findings")
       except Exception as e:
           logger.error(f"Code analysis {code_analysis_id} failed: {e}")
           if ca:
               ca.status = "failed"
               ca.error_message = str(e)[:1000]
               db.commit()
       finally:
           db.close()
   ```
